### **`yolo2coco.py` 转换脚本**

**使用说明：**
1.  将下面的代码保存为 `yolo2coco.py`，并放在你项目的主文件夹下（与 `dataset` 文件夹同级）。
2.  确保你的目录结构符合脚本预期。
3.  运行脚本：`python yolo2coco.py`
4.  脚本执行完毕后，会在 `dataset/` 目录下创建一个 `annotations` 文件夹，里面包含了 `instances_train.json`, `instances_val.json`, 和 `instances_test.json`。

```python
# -*- coding: utf-8 -*-
import os
import json
import cv2  # pip install opencv-python
from tqdm import tqdm # pip install tqdm

# --- 1. 配置区域 ---

# 数据集根目录
DATASET_BASE_DIR = "dataset"

# 需要转换的数据集划分
# 脚本会依次查找 dataset/images/train, dataset/images/val, dataset/images/test
SPLITS = ['train', 'val', 'test']

# 类别名称映射
CLASS_MAPPING = {
    0: 'dent',
    1: 'hole',
    2: 'rusty'
}

# 输出的 COCO 格式标注文件存放目录
OUTPUT_ANNOTATIONS_DIR = os.path.join(DATASET_BASE_DIR, "annotations")


# --- 2. 核心转换逻辑 ---

def convert_yolo_to_coco():
    """
    主函数，执行所有数据集划分的转换。
    """
    # 确保输出目录存在
    os.makedirs(OUTPUT_ANNOTATIONS_DIR, exist_ok=True)

    # 首先，构建 COCO 的 'categories' 字段
    # 注意：COCO 的 category_id 通常从 1 开始，0 一般留给背景。
    # 所以我们将 YOLO 的 class_id (0, 1, 2) 映射到 COCO 的 category_id (1, 2, 3)
    categories = []
    for yolo_id, name in CLASS_MAPPING.items():
        categories.append({
            "id": yolo_id + 1,  # COCO category id 从 1 开始
            "name": name,
            "supercategory": "defect"
        })

    # 遍历所有需要转换的数据集划分 (train, val, test)
    for data_split in SPLITS:
        print(f"\n--- 开始处理 {data_split.upper()} 数据集 ---")
        
        # 初始化当前划分的COCO数据结构
        coco_data = {
            "images": [],
            "annotations": [],
            "categories": categories
        }

        # 构建路径
        image_dir = os.path.join(DATASET_BASE_DIR, "images", data_split)
        label_dir = os.path.join(DATASET_BASE_DIR, "labels", data_split)
        
        if not os.path.exists(image_dir):
            print(f"警告: 目录 {image_dir} 不存在，跳过 {data_split} 数据集。")
            continue

        # 初始化ID计数器
        image_id_counter = 0
        annotation_id_counter = 0

        # 遍历图片文件夹
        image_filenames = [f for f in os.listdir(image_dir) if f.lower().endswith(('.jpg', '.png', '.jpeg'))]
        for image_name in tqdm(image_filenames, desc=f"Processing {data_split} images"):
            image_path = os.path.join(image_dir, image_name)

            # 读取图片尺寸
            try:
                img = cv2.imread(image_path)
                img_h, img_w, _ = img.shape
            except Exception as e:
                print(f"\n错误: 无法读取图片 {image_path}，已跳过。错误信息: {e}")
                continue

            # 添加 'images' 条目
            image_info = {
                "id": image_id_counter,
                "file_name": image_name,
                "width": img_w,
                "height": img_h
            }
            coco_data["images"].append(image_info)

            # 处理对应的标签文件
            label_name = os.path.splitext(image_name)[0] + ".txt"
            label_path = os.path.join(label_dir, label_name)

            if os.path.exists(label_path):
                with open(label_path, 'r') as f:
                    for line in f.readlines():
                        parts = line.strip().split()
                        if len(parts) != 5:
                            continue

                        yolo_class_id = int(parts[0])
                        x_center, y_center, width, height = map(float, parts[1:])

                        # 将 YOLO 归一化坐标转换为 COCO 像素坐标 [x_min, y_min, width, height]
                        coco_w = width * img_w
                        coco_h = height * img_h
                        coco_x = (x_center * img_w) - (coco_w / 2)
                        coco_y = (y_center * img_h) - (coco_h / 2)

                        # 添加 'annotations' 条目
                        annotation_info = {
                            "id": annotation_id_counter,
                            "image_id": image_id_counter,
                            "category_id": yolo_class_id + 1, # 对应 categories 中的 id
                            "bbox": [coco_x, coco_y, coco_w, coco_h],
                            "area": coco_w * coco_h,
                            "iscrowd": 0,
                            "segmentation": [] # 对于目标检测，此项可以为空
                        }
                        coco_data["annotations"].append(annotation_info)
                        annotation_id_counter += 1
            
            # 增加图片ID计数器
            image_id_counter += 1

        # 将当前划分的数据保存为 JSON 文件
        output_json_path = os.path.join(OUTPUT_ANNOTATIONS_DIR, f"instances_{data_split}.json")
        with open(output_json_path, 'w') as f:
            json.dump(coco_data, f, indent=4)
            
        print(f"{data_split.upper()} 数据集转换完成！")
        print(f"总计图片数: {len(coco_data['images'])}")
        print(f"总计标注框数: {len(coco_data['annotations'])}")
        print(f"标注文件已保存至: {output_json_path}")


# --- 3. 脚本执行入口 ---

if __name__ == "__main__":
    convert_yolo_to_coco()
    print("\n======================================")
    print("所有数据集格式转换已完成！✅")
    print("======================================")

```

**关键提醒：** 在修改配置文件时，请特别注意 `num_classes` 和 `category_id` 的对应关系。

*   脚本已经自动将YOLO的 `class_id` `0, 1, 2` 映射为了COCO的 `category_id` `1, 2, 3`。MMDetection能够正确处理这种从1开始的ID，你无需再做任何额外操作。
