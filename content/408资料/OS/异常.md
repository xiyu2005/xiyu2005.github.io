汇编代码段
### 场景一：系统调用（主动陷入内核）

这是用户态请求操作系统服务的唯一入口。

**x86 示例（Linux 经典写法）：**
```asm
; 功能：在终端打印字符 'A'
mov eax, 4          ; EAX=4 表示 sys_write 系统调用号
mov ebx, 1          ; EBX=1 表示标准输出（stdout）
mov ecx, char_A     ; ECX 指向要打印的字符地址
mov edx, 1          ; EDX 表示打印长度 1 字节
int 0x80            ; 执行软件中断，CPU 查 IDT[0x80]，陷入内核态
```

**MIPS 示例（MARS 模拟器经典写法）：**
```asm
; 功能：程序退出（终止运行）
li $v0, 10          ; $v0=10 表示 exit 系统调用号
syscall             ; 触发系统调用异常，CPU 跳转到 0x80000180（异常入口）
```

### 场景二：溢出主动捕获（条件异常）

对应你之前问的 `IMUL` + `INTO` 组合，以及 MIPS 的模拟实现。

**x86 示例：**
```asm
mov eax, 0x7FFFFFFF ; EAX = 最大正数 (2^31 - 1)
mov ecx, 2          
imul ecx            ; 计算 EAX * ECX，结果 = 0xFFFFFFFE 存入 EDX:EAX
                    ; 此时 EAX = 0xFFFFFFFE（负数），EDX = 0x00000000（未全F），OF=1
into                ; 检查 OF 标志。因为 OF=1，触发 4 号中断（溢出异常）
                    ; 如果 OF=0，这条指令相当于 NOP（空操作）
add eax, 1          ; 如果没溢出，程序继续往下走（本例不会执行到这里）
```

**MIPS 示例（虽无直接 INTO，但常用条件分支模拟）：**
```asm
; 假设 t0 和 t1 是乘数，mult 结果放入 hi, lo
mult $t0, $t1       ; 有符号乘，高32位存入 hi，低32位存入 lo
mfhi $t2            ; 将高32位（hi）取出到 t2
srl $t3, $lo, 31    ; 取出低32位（lo）的最高位（符号位）
bne $t2, $t3, overflow_handler ; 如果高位不等于低位的符号扩展，跳转溢出处理
```

### 场景三：调试断点（被动暂停）

**x86 示例：**
```asm
; 在程序关键位置插入断点
mov ax, data_seg    
mov ds, ax          
int 3               ; 机器码 0xCC，触发 3 号中断
                    ; CPU 执行到这里会暂停，把控制权交给调试器（如 GDB）
mov bx, 0x1234      ; 调试器让程序继续运行后，才会执行这里
```

**MIPS 示例：**
```asm
addi $t0, $zero, 5
break               ; 触发断点异常，CPU 跳转到异常处理程序（通常用于模拟器调试）
addi $t1, $zero, 10 ; 调试器恢复执行后才会运行这条
```

### 场景四：边界检查（数组越界检测）

**x86 专用指令 `BOUND`（408 冷门考点）：**
```asm
.data
array_bounds dw 0, 9    ; 定义数组索引的 下限（0） 和 上限（9）
index       dw 10       ; 当前要访问的下标

.code
mov bx, index           
bound bx, array_bounds  ; 检查 BX 是否在 [0, 9] 区间内
                        ; 因为 BX=10 大于上限 9，触发 5 号中断（边界检查异常）
mov ax, array[bx]       ; 如果没有越界，才执行这里读取数组
```

### 场景五：中断服务程序的收尾与嵌套控制

**x86 示例（ISR 标准模板）：**
```asm
; 假设这是时钟中断服务程序（IRQ 0）
clock_interrupt_handler:
    push eax            ; 保护现场（保存通用寄存器）
    push ebx
    
    sti                 ; 开中断！（允许响应更高优先级的外部中断，实现嵌套）
    
    ; ... 在这里执行具体的时钟中断处理逻辑（如更新系统时间） ...
    
    cli                 ; 关中断！（准备退出，保证恢复现场的几步不被干扰）
    pop ebx             ; 恢复现场
    pop eax
    iret                ; 中断返回（恢复 CS:EIP 和 EFLAGS，回到被中断的程序）
```

**MIPS 示例：**
```asm
; 异常处理程序入口
exception_handler:
    mfc0 $k0, $14       ; 从协处理器0取出异常返回地址（EPC）
    ; ... 处理异常逻辑 ...
    
    eret                ; 异常返回（将 EPC 写回 PC，并恢复用户态状态）
```

### 💡 408 考场终极记忆法
-   **`INT` / `syscall`**：程序主动敲门（用户态 -> 内核态）。
-   **`INTO` / `BOUND` / `break`**：程序主动设卡（检查条件，错了就跳）。
-   **`IRET` / `eret`**：程序退房结账（内核态 -> 用户态）。
-   **`STI` / `CLI`**：程序的门禁开关（允许/禁止外部中断打扰）。
