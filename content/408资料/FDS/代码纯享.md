合并两个无序链表
首先把链表B接到A尾部，合成一条完整的无序链表；
然后分，用快慢指针找到中点，切开为左右两段；
递归对左右两段归并排序，直到空或者仅有一个节点；
合并两段有序的子链表。
逐层返回。
```c
typedef struct LNode{
    int data;
    struct LNode *next;
}LNode,*LinkList;

// 归并两个有序链表
LNode* merge(LNode* pa, LNode* pb){
    LNode *C, *tail;
    C = (LNode*)malloc(sizeof(LNode));
    C->next = NULL; tail = C;
    while(pa && pb){
        if(pa->data < pb->data){
            tail->next = pa; pa = pa->next;
        }else{
            tail->next = pb; pb = pb->next;
        }
        tail = tail->next;
    }
    tail->next = pa?pa:pb;
    return C->next;
}

// 链表归并排序（自顶向下，快慢指针分割）
LNode* mergeSort(LNode* head){
    if(head==NULL || head->next==NULL) return head;
    LNode *slow=head, *fast=head;
    while(fast->next && fast->next->next){
        slow = slow->next;
        fast = fast->next->next;
    }
    LNode *mid = slow->next;
    slow->next = NULL;
    LNode *left = mergeSort(head);
    LNode *right = mergeSort(mid);
    return merge(left,right);
}

// 主函数：合并两条无序不带头链表，得到升序链表
LNode* MergeTwoUnorder(LinkList A, LinkList B){
    if(A == NULL) return mergeSort(B);
    LNode *p = A;
    while(p->next) p = p->next;
    p->next = B; // 拼接两条链表
    return mergeSort(A);
}
```

2016年算法题，堆解法/优先队列

```c
#include <stdio.h>
#include <stdlib.h>

void swap(int *a, int *b) {
    int t = *a; *a = *b; *b = t;
}

// 大顶堆下沉调整
void siftDown(int heap[], int size, int i) {
    int largest = i;
    int l = 2 * i + 1;
    int r = 2 * i + 2;
    if (l < size && heap[l] > heap[largest]) largest = l;
    if (r < size && heap[r] > heap[largest]) largest = r;
    if (largest != i) {
        swap(&heap[i], &heap[largest]);
        siftDown(heap, size, largest);
    }
}

// 对前 k 个元素建大顶堆
void buildHeap(int heap[], int k) {
    for (int i = k / 2 - 1; i >= 0; i--) {
        siftDown(heap, k, i);
    }
}

// 用 x 替换堆顶，并保持大顶堆性质
void replaceTop(int heap[], int k, int x) {
    heap[0] = x;
    siftDown(heap, k, 0);
}

int main() {
    int n;
    scanf("%d", &n);
    int *A = (int*)malloc(n * sizeof(int));
    for (int i = 0; i < n; i++) scanf("%d", &A[i]);

    int k = n / 2;                       // 较小集合的元素个数
    int *heap = (int*)malloc(k * sizeof(int));

    // 先取前 k 个元素放入堆
    for (int i = 0; i < k; i++) heap[i] = A[i];
    buildHeap(heap, k);                  // 建大顶堆

    // 遍历剩余元素，若小于堆顶则替换
    for (int i = k; i < n; i++) {
        if (A[i] < heap[0]) {
            replaceTop(heap, k, A[i]);
        }
    }

    // 堆中即为最小的 k 个元素
    long long S1 = 0, S2 = 0;
    for (int i = 0; i < k; i++) S1 += heap[i];
    for (int i = 0; i < n; i++) S2 += A[i];
    S2 -= S1;

    printf("n1 = %d, n2 = %d\n", k, n - k);
    printf("S1 = %lld, S2 = %lld\n", S1, S2);
    printf("|S1 - S2| = %lld\n", llabs(S1 - S2));

    free(A);
    free(heap);
    return 0;
}
```
