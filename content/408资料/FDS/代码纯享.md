合并两个无序链表
首先把链表B接到A尾部，合成一条完整的无序链表；
然后分，用快慢指针找到中点，切开为左右两段；
递归对左右两段归并排序，直到空或者仅有一个节点；
合并两段有序的子链表。
逐层返回。
```c
typedef struct LNode{
    int data;
    struct LNode *next;
}LNode,*LinkList;

// 归并两个有序链表
LNode* merge(LNode* pa, LNode* pb){
    LNode *C, *tail;
    C = (LNode*)malloc(sizeof(LNode));
    C->next = NULL; tail = C;
    while(pa && pb){
        if(pa->data < pb->data){
            tail->next = pa; pa = pa->next;
        }else{
            tail->next = pb; pb = pb->next;
        }
        tail = tail->next;
    }
    tail->next = pa?pa:pb;
    return C->next;
}

// 链表归并排序（自顶向下，快慢指针分割）
LNode* mergeSort(LNode* head){
    if(head==NULL || head->next==NULL) return head;
    LNode *slow=head, *fast=head;
    while(fast->next && fast->next->next){
        slow = slow->next;
        fast = fast->next->next;
    }
    LNode *mid = slow->next;
    slow->next = NULL;
    LNode *left = mergeSort(head);
    LNode *right = mergeSort(mid);
    return merge(left,right);
}

// 主函数：合并两条无序不带头链表，得到升序链表
LNode* MergeTwoUnorder(LinkList A, LinkList B){
    if(A == NULL) return mergeSort(B);
    LNode *p = A;
    while(p->next) p = p->next;
    p->next = B; // 拼接两条链表
    return mergeSort(A);
}
```
